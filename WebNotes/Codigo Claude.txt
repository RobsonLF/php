<script>
  // Timer simples para a Sessão de Foco (25 min - técnica pomodoro)
  const focusBtn = document.getElementById('focusBtn');
  const focusLabel = document.getElementById('focusLabel');
  let timerInterval = null;
  let secondsLeft = 25 * 60;
  let running = false;

  function formatTime(s){
    const m = Math.floor(s/60).toString().padStart(2,'0');
    const sec = (s%60).toString().padStart(2,'0');
    return `${m}:${sec}`;
  }

  focusBtn.addEventListener('click', () => {
    if(!running){
      running = true;
      timerInterval = setInterval(() => {
        secondsLeft--;
        focusLabel.textContent = formatTime(secondsLeft);
        if(secondsLeft <= 0){
          clearInterval(timerInterval);
          running = false;
          focusLabel.textContent = 'Sessão concluída!';
        }
      }, 1000);
    } else {
      running = false;
      clearInterval(timerInterval);
      focusLabel.textContent = 'Pausado — Retomar';
    }
  });

  // Feedback visual ao clicar em "Adicionar Novo Curso"
  document.getElementById('addCourseBtn').addEventListener('click', () => {
    alert('Aqui você abriria o formulário de novo curso.');
  });
